    <script>
        var app = angular.module("shopApp", ["ngAnimate"]);

        // CUSTOM SERVICE 1: ProductService
        app.service("ProductService", function() {
            var products = [
                {name: "Laptop", category: "ELECTRONICS", price: 50000,
                 description: "A laptop for study, work and entertainment"},
                {name: "Headphones", category: "ELECTRONICS", price: 1500,
                 description: "Comfortable headphones with clear sound"},
                {name: "Backpack", category: "ACCESSORIES", price: 1200,
                 description: "A useful backpack for college students"}
            ];

            this.getProducts = function() {
                return products;
            };
        });

        // CUSTOM SERVICE 2: CartService
        app.service("CartService", function() {
            var cart = [];

            this.getItems = function() {
                return cart;
            };

            this.addItem = function(product) {
                cart.push(product);
            };

            this.removeItem = function(index) {
                cart.splice(index, 1);
            };

            this.getTotal = function() {
                var total = 0;
                for (var i = 0; i < cart.length; i++) {
                    total += cart[i].price;
                }
                return total;
            };
        });

        // CUSTOM FILTER 1: capitalize
        app.filter("capitalize", function() {
            return function(input) {
                if (!input) return "";
                return input.charAt(0).toUpperCase() + input.slice(1).toLowerCase();
            };
        });

        // CUSTOM FILTER 2: truncate
        app.filter("truncate", function() {
            return function(input, limit) {
                if (!input) return "";
                return input.length > limit
                    ? input.substring(0, limit) + "..."
                    : input;
            };
        });

        app.controller("ShopController", function(
            $scope, $http, $timeout, ProductService, CartService
        ) {

            // Built-in service 2: $timeout (clears feedback after 2 seconds)
            $scope.products = ProductService.getProducts();
            $scope.cart = CartService.getItems();
            $scope.searchText = "";
            $scope.sortOption = "name";
            $scope.message = "";

            $scope.addToCart = function(product) {
                CartService.addItem(product);
                $scope.message = product.name + " added to cart!";
                $timeout(function() {
                    $scope.message = "";
                }, 2000);
            };

            $scope.removeFromCart = function(index) {
                CartService.removeItem(index);
            };

            $scope.getTotal = function() {
                return CartService.getTotal();
            };

            // Built-in service 1: $http
            $scope.loadFromAPI = function() {
                $http.get("https://dummyjson.com/products?limit=6")
                    .then(function(response) {
                        $scope.products = response.data.products.slice(0, 6).map(function(p) {
                            return {
                                name: p.title,
                                category: p.category,
                                price: p.price,
                                description: p.description
                            };
                        });
                        $scope.message = "6 products loaded from API!";
                    })
                    .catch(function() {
                        $scope.message = "Could not load API products.";
                    });
            };

            $scope.checkout = function() {
                $scope.message = "Checkout completed for demo!";
            };
        });
    </script>
</body>

```python
from flask import Flask, request, render_template_string
from pymongo import MongoClient

app = Flask(__name__)

client = MongoClient("mongodb+srv://USERNAME:PASSWORD@cluster0.4drwvvj.mongodb.net/?appName=Cluster0")
db = client["student"]
records = db["students"]

html = """
<h2>Student Form</h2>
<form method="post">
Name: <input name="name" required><br><br>
Email: <input name="email" type="email" required><br><br>
Phone: <input name="phone" required><br><br>
<button name="action" value="Create">Create</button>
<button name="action" value="Update">Update</button>
<button name="action" value="Delete">Delete</button>
</form>

<form method="get">
<button>View All</button>
</form>

<h2>Student Records</h2>
<table border="1" cellpadding="10">
<tr><th>Name</th><th>Email</th><th>Phone</th></tr>
{% for s in students %}
<tr>
<td>{{ s.name }}</td>
<td>{{ s.email }}</td>
<td>{{ s.phone }}</td>
</tr>
{% endfor %}
</table>
<p>{{ msg }}</p>
"""

@app.route("/", methods=["GET", "POST"])
def home():
    msg = ""

    if request.method == "POST":
        name = request.form["name"]
        email = request.form["email"]
        phone = request.form["phone"]
        action = request.form["action"]

        if action == "Create":
            records.insert_one({
                "name": name, "email": email, "phone": phone
            })
            msg = "Student added!"

        elif action == "Update":
            result = records.update_one(
                {"email": email},
                {"$set": {"name": name, "phone": phone}}
            )
            msg = "Updated!" if result.matched_count else "Email not found!"

        elif action == "Delete":
            result = records.delete_one({"email": email})
            msg = "Deleted!" if result.deleted_count else "Email not found!"

    students = list(records.find({}, {"_id": 0}))
    return render_template_string(html, students=students, msg=msg)

if __name__ == "__main__":
    app.run(debug=True)
```

    <input id="password" name="password"
           type="password" placeholder="Password" required><br><br>

<p id="message">{{ message }}</p>
"""

@app.route("/", methods=["GET", "POST"])
def login():
    message = ""

    if request.method == "POST":
        username = request.form.get("username")
        password = request.form.get("password")

        if username == "admin" and password == "12345":
            message = "Login Successful"
        else:
            message = "Invalid username or password"

    return render_template_string(HTML, message=message)

if __name__ == "__main__":

from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

driver = webdriver.Chrome()
wait = WebDriverWait(driver, 10)

try:
    # POSITIVE TEST CASE
    driver.get("http://127.0.0.1:5000")

    driver.find_element(By.ID, "username").send_keys("admin")
    driver.find_element(By.ID, "password").send_keys("12345")
    driver.find_element(By.ID, "loginBtn").click()

    message = wait.until(
        EC.visibility_of_element_located((By.ID, "message"))
    ).text

    if message == "Login Successful":
        print("Positive Test: PASS")
    else:
        print("Positive Test: FAIL")

    # NEGATIVE TEST CASE
    driver.get("http://127.0.0.1:5000")

    driver.find_element(By.ID, "username").send_keys("admin")
    driver.find_element(By.ID, "password").send_keys("wrong123")
    driver.find_element(By.ID, "loginBtn").click()

    message = wait.until(
        EC.visibility_of_element_located((By.ID, "message"))
    ).text

    if message == "Invalid username or password":
        print("Negative Test: PASS")
    else:
        print("Negative Test: FAIL")

finally:
    driver.quit()
