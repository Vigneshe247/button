<script>
var app = angular.module('shop', ['ngAnimate']);

app.factory('DataFactory', function($q) {
  var data = [
    {id:1, name:'Laptop', amount:78000, details:'16GB RAM'},
    {id:2, name:'Phone', amount:25000, details:'8GB RAM'},
    {id:3, name:'Headphones', amount:2000, details:'Wireless'},
    {id:4, name:'Keyboard', amount:1500, details:'Mechanical'}
  ];
  return {load:function() { return $q.when(angular.copy(data)); }};
});

app.service('ShopService', function() {
  this.items = {cart:[], liked:[]};
  this.nextKey = 1;

  this.add = function(product, type) {
    if (!product || !product.name || !product.amount ||
        Number(product.amount) <= 0 || !this.items[type]) return 'invalid';

    var name = product.name.trim().toLowerCase();
    var exists = this.items[type].some(function(item) {
      if (product.id !== undefined && product.id !== null)
        return item.id === product.id;
      return item.name.trim().toLowerCase() === name;
    });
    if (exists) return 'exists';

    var item = angular.copy(product);
    item.name = item.name.trim();
    item.amount = Number(item.amount);
    item.key = this.nextKey++;
    this.items[type].push(item);
    return 'added';
  };
});

app.directive('box', function() {
  return {restrict:'C', link:function(scope, element) {
    element.css('border-radius', '8px');
  }};
});

app.controller('ShopCtrl', function($scope, DataFactory, ShopService) {
  $scope.items = ShopService.items;
  $scope.products = [];
  $scope.view = 'cart';
  $scope.custom = {type:'cart'};
  $scope.search = '';
  $scope.amountQuery = '';
  $scope.amountResult = null;
  $scope.searched = false;
  $scope.msg = '';

  DataFactory.load().then(function(data) { $scope.products = data; });

  $scope.add = function(product, type) {
    var result = ShopService.add(product, type);
    $scope.msg = result === 'added' ? 'Added successfully!' :
      result === 'exists' ? 'This product is already in this list.' :
      'Enter a name and a valid amount greater than zero.';
  };

  $scope.addCustom = function() {
    $scope.add($scope.custom, $scope.custom.type);
    $scope.custom = {type:'cart'};
  };

  $scope.showItems = function(type) { $scope.view = type; };

  $scope.remove = function(product, type) {
    var list = $scope.items[type];
    var index = list.indexOf(product);
    if (index !== -1) list.splice(index, 1);
    if ($scope.amountResult && $scope.amountResult.name.toLowerCase() === product.name.toLowerCase()) {
      $scope.amountResult = null;
      $scope.searched = false;
    }
  };

  $scope.total = function() {
    return $scope.items.cart.reduce(function(sum, p) {
      return sum + Number(p.amount);
    }, 0);
  };

  $scope.searchAmount = function() {
    $scope.searched = true;
    $scope.amountResult = null;
    var query = ($scope.amountQuery || '').trim().toLowerCase();
    if (!query) return;

    var found = $scope.items.cart.concat($scope.items.liked).find(function(p) {
      return p.name.trim().toLowerCase() === query;
    });
    if (found) $scope.amountResult = found;
  };
});
</script>

@app.route("/", methods=["GET", "POST"])
def home():
    msg = ""

    if request.method == "POST":
        name = request.form["name"]
        email = request.form["email"]
        phone = request.form["phone"]
        action = request.form["action"]

        if action == "Create":
            records.insert_one({
                "name": name, "email": email, "phone": phone
            })
            msg = "Student added!"

        elif action == "Update":
            result = records.update_one(
                {"email": email},
                {"$set": {"name": name, "phone": phone}}
            )
            msg = "Updated!" if result.matched_count else "Email not found!"

        elif action == "Delete":
            result = records.delete_one({"email": email})
            msg = "Deleted!" if result.deleted_count else "Email not found!"

    students = list(records.find({}, {"_id": 0}))
    return render_template_string(html, students=students, msg=msg)

if __name__ == "__main__":
    app.run(debug=True)
